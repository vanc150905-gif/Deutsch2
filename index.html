    function recordAttempt(id, isCorrect) {

      const now = Date.now();

      const record = getWordSrs(id);

      if (record.lastReviewed && (now - record.lastReviewed < 4000) && record.lastResult === isCorrect) {

        return;

      }

      record.lastReviewed = now;

      record.lastResult = isCorrect;

      if (isCorrect) {

        record.correctCount += 1;

        record.streak += 1;

        const intervalIdx = Math.min(record.streak, SRS_INTERVALS_DAYS.length - 1);

        const days = SRS_INTERVALS_DAYS[intervalIdx];

        record.nextReview = now + days * DAY_MS;

      } else {

        record.wrongCount += 1;

        record.streak = 0;

        record.nextReview = now;

      }

      srsData[id] = record;

      saveSrs();

    }

    function insertUmlaut(event, char) {

      event.preventDefault();

      if (!lastFocusedInput) return;

      const start = lastFocusedInput.selectionStart || 0;

      const end = lastFocusedInput.selectionEnd || 0;

      const val = lastFocusedInput.value;

      lastFocusedInput.value = val.slice(0, start) + char + val.slice(end);

      lastFocusedInput.focus();

      lastFocusedInput.setSelectionRange(start + char.length, start + char.length);

      const idMatch = lastFocusedInput.id.match(/^input-(\d+)$/);

      if (idMatch) {

        handleInput(Number(idMatch[1]), lastFocusedInput.value);

      }

    }

    function focusNextIncompleteRow(currentId) {

      const inputs = Array.from(document.querySelectorAll('#vocab-tbody input[id^="input-"]'));

      const currentIndex = inputs.findIndex(el => el.id === `input-${currentId}`);

      if (currentIndex === -1) return;

      // Look forward for the next unanswered or wrong row

      for (let i = currentIndex + 1; i < inputs.length; i++) {

        const nextId = Number(inputs[i].id.replace("input-", ""));

        const nextItem = vocabList.find(v => v.id === nextId);

        if (nextItem && checkAnswer(nextItem, userAnswers[nextId] || "") !== "CORRECT") {

          inputs[i].focus();

          inputs[i].scrollIntoView({ behavior: "smooth", block: "center" });

          return;

        }

      }

      // Wrap around from the top if there are earlier incomplete rows

      for (let i = 0; i < currentIndex; i++) {

        const nextId = Number(inputs[i].id.replace("input-", ""));

        const nextItem = vocabList.find(v => v.id === nextId);

        if (nextItem && checkAnswer(nextItem, userAnswers[nextId] || "") !== "CORRECT") {

          inputs[i].focus();

          inputs[i].scrollIntoView({ behavior: "smooth", block: "center" });

          return;

        }

      }

    }

    function speakGerman(id, mode) {

      const textToSpeak = mode === "word" ? item.de : item.ex;

      const utterance = new SpeechSynthesisUtterance(textToSpeak);

      utterance.lang = "de-DE";

      utterance.rate = speechRate;

      const voices = window.speechSynthesis.getVoices();

      const germanVoice = voices.find(v => v.lang.startsWith("de"));

      if (germanVoice) {

        utterance.voice = germanVoice;

      }

      window.speechSynthesis.speak(utterance);

    }

    if ("speechSynthesis" in window) {

      window.speechSynthesis.onvoiceschanged = () => window.speechSynthesis.getVoices();

    }

    function handleBulkImport() {

      const raw = document.getElementById("bulk-textarea").value.trim();

      if (!raw) return;

      const lines = raw.split(/

?

/);

      let maxId = vocabList.reduce((max, item) => Math.max(max, item.id), 0);

      let addedCount = 0;

      let lastLektion = "Lektion mới";

      lines.forEach(line => {

        if (!line.trim()) return;

        const parts = line.includes("        ") ? line.split("        ") : line.split("|");

        if (parts.length >= 4) {

          maxId++;

          const lektion = parts[0].trim() || "Lektion mới";

          const posRaw = parts[1].trim();

          const pos = ["N", "V", "Adj"].includes(posRaw) ? posRaw : "N";

          const vi = parts[2].trim();

          const de = parts[3].trim();

          const ex = (parts[4] || "").trim();

          const ipa = (parts[5] || "").trim();

          const grammar = (parts[6] || "").trim();

          const clozeTarget = de.replace(/^(der|die|das|sich)\s+/i, "").trim();

          vocabList.unshift({ id: maxId, lektion, pos, vi, de, clozeTarget, ex, ipa, grammar });

          lastLektion = lektion;

          addedCount++;

        }

      });

      if (addedCount > 0) {

        saveVocab();

        populateLektionDropdown();

        document.getElementById("filter-lektion").value = lastLektion;

        renderTable();

        document.getElementById("bulk-textarea").value = "";

        closeAddModal();

        alert(`Đã nạp thành công ${addedCount} từ vựng vào ${lastLektion}!`);

      } else {

        alert("Không nhận diện được dòng hợp lệ. Vui lòng kiểm tra định dạng phân cách bởi dấu Tab hoặc dấu |.");

      }

    }
